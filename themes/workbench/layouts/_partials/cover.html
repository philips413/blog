{{- /* 글 목록용 대표 이미지(작은 미리보기). 인자: 글 페이지. 이미지가 없으면 아무것도 출력하지 않는다. 고르는 순서: 머리말 image → 본문 첫 이미지(마크다운·HTML 모두, 렌더 결과 기준). assets/ 또는 글 번들에 있는 이미지는 미리보기 크기로 잘라 작은 파일을 만들고, 외부 주소는 원본을 줄여 보여준다. */ -}} {{- $p := . -}} {{- $src := "" -}} {{- with $p.Params.image -}} {{- $src = . -}} {{- else -}} {{- with findRESubmatch `]*?\ssrc="([^"]+)"` $p.Content 1 -}} {{- $src = index (index . 0) 1 | htmlUnescape -}} {{- end -}} {{- end -}} {{- with $src -}} {{- $u := urls.Parse . -}} {{- $res := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "/" $u.Path -}} {{- with or (resources.Get $path) ($p.Resources.Get $path) -}} {{- $res = . -}} {{- else -}} {{- warnf "대표 이미지를 찾을 수 없음: %q (%s) — assets/%s 에 파일이 있는지 확인하세요." $src $p.File.Path $path -}} {{- end -}} {{- end -}} {{- if $res -}} {{- if and (eq $res.MediaType.MainType "image") (not (in (slice "svg" "gif") $res.MediaType.SubType)) -}} {{- $t := $res.Fill "144x108 Center q80" -}} {{- else -}} {{- end -}} {{- else if $u.IsAbs -}} {{- end -}} {{- end -}}
